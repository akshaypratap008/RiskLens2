Profitability:
ebitda_margin = ebitda / revenue
net_margin = net_profit / revenue

Liquidity:
current_ratio = current_assets / current_liabilities

quick_ratio = (
    current_assets - inventory
) / current_liabilities

working_capital = (
    current_assets
    - current_liabilities
)

Leverage:
debt_to_ebitda = (
    total_debt / ebitda
)

debt_to_equity = (
    total_debt / equity
)

net_debt_to_ebitda = (
    total_debt - cash
) / ebitda

Coverage:
interest_cover = (
    ebitda / interest_expense
)
